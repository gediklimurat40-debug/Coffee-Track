<script>
    const customers = [
        { avatar: '🧔', name: 'Hipster Alex' },
        { avatar: '👩‍🦰', name: 'Student Sarah' },
        { avatar: '👨‍💼', name: 'Manager Bob' },
        { avatar: '👵', name: 'Granny Emma' }
    ];

    const menu = ['Espresso', 'Iced Latte', 'Croissant', 'Donut'];

    let score = 0;
    let timeLeft = 60;
    let currentOrder = '';
    let gameActive = true;

    function generateCustomer() {
        if (!gameActive) return;
        const randomCustomer = customers[Math.floor(Math.random() * customers.length)];
        currentOrder = menu[Math.floor(Math.random() * menu.length)];
        
        document.getElementById('customer-avatar').innerText = randomCustomer.avatar;
        document.getElementById('customer-name').innerText = randomCustomer.name;
        document.getElementById('order-item').innerText = currentOrder;
    }

    function showMessage(text) {
        const msgEl = document.getElementById('msg');
        msgEl.innerText = text;
        msgEl.style.display = 'block';
        setTimeout(() => { msgEl.style.display = 'none'; }, 800);
    }

    function serveOrder(item) {
        if (!gameActive) return;

        if (item === currentOrder) {
            score += 10; // +10 points for correct order
            showMessage('+10 Perfect!');
        } else {
            score = Math.max(0, score - 5); // -5 points penalty for wrong order
            showMessage('-5 Wrong Item!');
        }

        document.getElementById('score').innerText = score;
        generateCustomer();
    }

    // Timer logic
    const timerInterval = setInterval(() => {
        if (timeLeft > 0) {
            timeLeft--;
            document.getElementById('timer').innerText = timeLeft;
        } else {
            gameActive = false;
            clearInterval(timerInterval);
            alert(`Game Over! Final Score: ${score}`);
        }
    }, 1000);

    // Initial Start
    generateCustomer();
</script>
